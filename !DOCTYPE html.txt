        <input
          type="text"
          id="keyword"
          placeholder="러닝 장소를 검색하세요"

        <button onclick="searchPlaces()">검색</button>

  <!-- 카카오맵 SDK -->
  <script
    type="text/javascript"
    src="https://dapi.kakao.com/v2/maps/sdk.js?appkey=1e75ade410bd5aa93f0e33ea5cffbb8c&libraries=services&autoload=false">
  </script>

  <script>

    let map;
    let currentMarker;
    let placesService;

    // 검색 마커 저장 배열
    let markers = [];

    // 인포윈도우 저장
    let infoWindows = [];

    // 카카오맵 로드 완료 후 실행
    kakao.maps.load(() => {

      navigator.geolocation.getCurrentPosition(

        // 위치 허용 성공
        (position) => {

          const latitude = position.coords.latitude;
          const longitude = position.coords.longitude;

          initializeMap(latitude, longitude);

        },

        // 위치 허용 실패
        () => {

          alert("위치 권한이 거부되었습니다.");

          // 기본 위치
          initializeMap(37.5665, 126.9780);

        }

      );

    });

    // 지도 초기화 함수
    function initializeMap(lat, lng) {

      const mapContainer = document.getElementById("map");

      const mapOption = {
        center: new kakao.maps.LatLng(lat, lng),
        level: 5
      };

      // 지도 생성
      map = new kakao.maps.Map(mapContainer, mapOption);

      // 장소 검색 서비스 생성
      placesService = new kakao.maps.services.Places();

      // 현재 위치 마커
      currentMarker = new kakao.maps.Marker({
        map: map,
        position: new kakao.maps.LatLng(lat, lng)
      });

      // 현재 위치 인포윈도우
      const currentInfo = new kakao.maps.InfoWindow({
        content: "<div style='padding:5px;'>📍 현재 위치</div>"
      });

      currentInfo.open(map, currentMarker);

    }

    // 검색 함수
    function searchPlaces() {

      const keyword = document.getElementById("keyword").value;

      // 공백 검사
      if (!keyword.trim()) {

        alert("검색어를 입력하세요.");
        return;

      }

      // 기존 검색 결과 제거
      clearSearchResults();

      // 현재 지도 중심 기준 검색
      const options = {
        location: map.getCenter(),
        radius: 5000
      };

      // 키워드 검색 실행
      placesService.keywordSearch(
        keyword,
        placesSearchCallback,
        options
      );

    }

    // 검색 결과 콜백
    function placesSearchCallback(data, status) {

      if (status === kakao.maps.services.Status.OK) {

        displayPlaces(data);

      } else {

        alert("검색 결과가 없습니다.");

      }

    }

    // 검색 결과 표시 함수
    function displayPlaces(places) {

      const listElement = document.getElementById("placesList");

      const bounds = new kakao.maps.LatLngBounds();

      places.forEach((place) => {

        const position = new kakao.maps.LatLng(place.y, place.x);

        // 지도 범위 확장
        bounds.extend(position);

        // 귀여운 이모지 마커 생성
        const customOverlay = new kakao.maps.CustomOverlay({
          position: position,
          content: `
            <div class="emoji-marker">
              🏃
            </div>
          `,
          yAnchor: 1
        });

        customOverlay.setMap(map);

        markers.push(customOverlay);

        // 랜덤 이미지 생성
        const imageUrl =
          "https://picsum.photos/220/120?random=" + Math.floor(Math.random() * 1000);

        // 인포윈도우 내용
        const infoContent = `
          <div class="info-window">
            <img src="${imageUrl}" alt="장소 이미지">
            <div class="info-title">${place.place_name}</div>
            <div class="info-address">${place.address_name}</div>
          </div>
        `;

        const infoWindow = new kakao.maps.InfoWindow({
          content: infoContent
        });

        infoWindows.push(infoWindow);

        // 오버레이 클릭 이벤트
        kakao.maps.event.addListener(map, "click", () => {
          infoWindow.close();
        });

        // 커스텀 오버레이 클릭 처리
        const markerElement = customOverlay.a;

        markerElement.addEventListener("click", () => {

          closeAllInfoWindows();

          infoWindow.open(map, new kakao.maps.Marker({
            position: position
          }));

          map.setCenter(position);

        });

        // 리스트 아이템 생성
        const item = document.createElement("li");

        item.innerHTML = `
          <div class="place-name">
            🏃 ${place.place_name}
          </div>

          <div class="place-address">
            📍 ${place.address_name}
          </div>
        `;

        // 리스트 클릭 이벤트
        item.addEventListener("click", () => {

          closeAllInfoWindows();

          infoWindow.open(map, new kakao.maps.Marker({
            position: position
          }));

          map.setCenter(position);

        });

        listElement.appendChild(item);

      });

      // 지도 범위 자동 조정
      map.setBounds(bounds);

    }

    // 기존 검색 결과 제거 함수
    function clearSearchResults() {

      // 기존 리스트 제거
      document.getElementById("placesList").innerHTML = "";

      // 기존 마커 제거
      markers.forEach((marker) => {
        marker.setMap(null);
      });

      markers = [];

      // 기존 인포윈도우 닫기
      closeAllInfoWindows();

    }

    // 인포윈도우 전체 닫기
    function closeAllInfoWindows() {

      infoWindows.forEach((infoWindow) => {
        infoWindow.close();
      });

      infoWindows = [];

    }

  </script>
